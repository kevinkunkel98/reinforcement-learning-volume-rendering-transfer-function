# Chat/Voice UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A local FastAPI web UI — chat (text + mic via Whisper) driving the
transfer function, rendered image display, back/forward history navigation, and
human-judgment buttons for hill-climbing — reusing the existing pipeline modules
unchanged.

**Architecture:** `hill_climb`'s pure search-step math is extracted out of
`mvp.py` into `search.py` so both `mvp.py` and the new `server.py` share it without
duplication. `server.py` (FastAPI) holds an in-memory `{history, cursor}` session
mirrored to `out/ui_session.json`, and exposes the endpoints from the design doc.
`static/index.html` + `app.js` + `style.css` is a plain no-build-step frontend.

**Tech Stack:** `fastapi`, `uvicorn`, `python-multipart` (new deps). Everything
else (`phantom`, `transfer`, `render`, `commands`, `evaluate`, `asr`) is reused as
committed.

Design reference: `docs/superpowers/specs/2026-09-02-chat-ui-design.md`.

```python
"""Pure hill-climbing search-step math, shared by mvp.py and server.py."""
import numpy as np

from transfer import PARAMS_PER_PEAK


def propose_step(params: np.ndarray, peak_idx: int, sign: float, step: float) -> np.ndarray:
    proposed = params.copy()
    base = peak_idx * PARAMS_PER_PEAK
    internal = (proposed[base + 2] + 1.0) / 2.0
    internal = float(np.clip(internal + sign * step, 0.0, 1.0))
    proposed[base + 2] = internal * 2.0 - 1.0
    return proposed


def resize_step(step: float, accepted: bool) -> float:
    return min(step * 1.2, 1.0) if accepted else step * 0.5
```

```python
"""FastAPI backend for the chat/voice UI."""
import base64
import datetime
import io
import json
import os
import uuid

import numpy as np
from fastapi import FastAPI, HTTPException
from fastapi.responses import FileResponse
from fastapi.staticfiles import StaticFiles
from pydantic import BaseModel

from phantom import build_phantom
from transfer import default_params, opacity_mass, TISSUE_BANDS
from render import render, grab, features
from commands import parse_command, apply_command
from evaluate import jsonl_append

UI_SESSION_PATH = os.environ.get("UI_SESSION_PATH", "out/ui_session.json")
LOG_PATH = "out/log.jsonl"

app = FastAPI()
_phantom = None
_history = []
_cursor = 0


def get_phantom():
    global _phantom
    if _phantom is None:
        _phantom = build_phantom()
    return _phantom


def _masses(params):
    return {t: opacity_mass(params, lo, hi) for t, (lo, hi) in TISSUE_BANDS.items()}


def _render_step(params, cmd_text, cmd_dict, verdict, search, step_id):
    img = grab(render(get_phantom(), params))
    from PIL import Image
    buf = io.BytesIO()
    Image.fromarray(img).save(buf, format="PNG")
    image_b64 = base64.b64encode(buf.getvalue()).decode()
    return {
        "id": step_id,
        "timestamp": datetime.datetime.now().isoformat(),
        "cmd_text": cmd_text,
        "cmd_dict": cmd_dict,
        "params": params.tolist(),
        "image_b64": image_b64,
        "masses": _masses(params),
        "features": features(img),
        "verdict": verdict,
        "search": search,
    }, img


def _save_session():
    os.makedirs(os.path.dirname(UI_SESSION_PATH) or ".", exist_ok=True)
    with open(UI_SESSION_PATH, "w") as f:
        json.dump({"history": _history, "cursor": _cursor}, f)


def _load_or_init_session():
    global _history, _cursor
    if os.path.exists(UI_SESSION_PATH):
        with open(UI_SESSION_PATH) as f:
            data = json.load(f)
        _history, _cursor = data["history"], data["cursor"]
        return
    step, _ = _render_step(default_params(), None, None, None, False, 0)
    _history = [step]
    _cursor = 0
    _save_session()


_load_or_init_session()
if os.path.isdir("static"):
    app.mount("/static", StaticFiles(directory="static"), name="static")


@app.get("/")
def index():
    return FileResponse("static/index.html")


@app.get("/api/state")
def get_state():
    return {"cursor": _cursor, "total": len(_history), "current": _history[_cursor], "pending": None}


@app.post("/api/back")
def back():
    global _cursor
    _cursor = max(0, _cursor - 1)
    _save_session()
    return get_state()


@app.post("/api/forward")
def forward():
    global _cursor
    _cursor = min(len(_history) - 1, _cursor + 1)
    _save_session()
    return get_state()


class CommandRequest(BaseModel):
    text: str
    parser: str = "rule"
    model: str = "qwen2.5:7b"
    search: bool = False
    evaluator: str = "objective"
    steps: int = 10


@app.post("/api/command")
def command(req: CommandRequest):
    global _history, _cursor
    try:
        cmd = parse_command(req.text, parser=req.parser, model=req.model)
    except ValueError as exc:
        raise HTTPException(status_code=400, detail=str(exc))

    current_params = np.array(_history[_cursor]["params"], dtype=np.float64)
    new_params = apply_command(cmd, current_params)
    step, _ = _render_step(new_params, req.text, cmd, None, False, _history[-1]["id"] + 1)

    _history = _history[:_cursor + 1] + [step]
    _cursor = len(_history) - 1
    jsonl_append(LOG_PATH, {
        "timestamp": step["timestamp"], "command": cmd,
        "params_before": current_params.tolist(), "params_after": new_params.tolist(),
        "features_before": _history[_cursor - 1]["features"], "features_after": step["features"],
        "verdict": None,
    })
    _save_session()
    return get_state()
```

```python
from search import propose_step, resize_step
from commands import _find_or_create_peak, STRENGTH_WORDS
from evaluate import objective

_pending = None  # dict holding an in-flight human-judged search, or None


def _run_objective_search(cmd_text, cmd, params, steps):
    sign = 1.0 if cmd["direction"] == "increase" else -1.0
    step = STRENGTH_WORDS[cmd["strength"] or "moderately"]
    current = params.copy()
    _, idx = _find_or_create_peak(current, cmd["target"])
    for _ in range(steps):
        proposed = propose_step(current, idx, sign, step)
        verdict = objective(current, proposed, cmd)
        jsonl_append(LOG_PATH, {
            "timestamp": datetime.datetime.now().isoformat(), "command": cmd,
            "params_before": current.tolist(), "params_after": proposed.tolist(),
            "features_before": None, "features_after": None, "verdict": verdict,
        })
        if verdict == 1:
            current = proposed
        step = resize_step(step, accepted=(verdict == 1))
        if step < 0.01:
            break
    return current
```

```python
@app.post("/api/command")
def command(req: CommandRequest):
    global _history, _cursor, _pending
    try:
        cmd = parse_command(req.text, parser=req.parser, model=req.model)
    except ValueError as exc:
        raise HTTPException(status_code=400, detail=str(exc))

    current_params = np.array(_history[_cursor]["params"], dtype=np.float64)

    if req.search and cmd["direction"] in ("increase", "decrease"):
        if req.evaluator == "human":
            _pending = _start_human_search(req.text, cmd, current_params, req.steps)
```

```python
from fastapi import UploadFile, File
from asr import _transcribe_path as asr_transcribe_path

AUDIO_DIR = "out/audio"


@app.post("/api/transcribe")
async def transcribe(audio: UploadFile = File(...)):
    os.makedirs(AUDIO_DIR, exist_ok=True)
    stamp = datetime.datetime.now().strftime("%Y%m%d-%H%M%S")
    wav_path = os.path.join(AUDIO_DIR, f"{stamp}.wav")
    with open(wav_path, "wb") as f:
        f.write(await audio.read())

    import time
    start = time.time()
    text = asr_transcribe_path(wav_path, lang="en", model_size="small")
    duration = time.time() - start

    with open(os.path.join(AUDIO_DIR, f"{stamp}.txt"), "w") as f:
        f.write(text)
    return {"text": text, "duration": duration}
```

- [ ] **Step 3: `static/app.js`**

```javascript
const state = { current: null, cursor: 0, total: 1, pending: null };

const el = (id) => document.getElementById(id);
const singleView = el("single-view");
const judgeView = el("judge-view");

async function refresh(data) {
  state.current = data.current;
  state.cursor = data.cursor;
  state.total = data.total;
  state.pending = data.pending;

  el("step-counter").textContent = `${state.cursor + 1} / ${state.total}`;
  el("back-btn").disabled = state.cursor <= 0;
  el("forward-btn").disabled = state.cursor >= state.total - 1;

  if (state.pending) {
    singleView.hidden = true;
    judgeView.hidden = false;
    el("before-image").src = `data:image/png;base64,${state.pending.before_image_b64}`;
    el("after-image").src = `data:image/png;base64,${state.pending.after_image_b64}`;
  } else {
    singleView.hidden = false;
    judgeView.hidden = true;
    el("current-image").src = `data:image/png;base64,${state.current.image_b64}`;
  }
}

function appendMessage(step) {
  if (!step.cmd_text) return;
  const div = document.createElement("div");
  div.className = "msg";
  const verdictText = step.verdict === 1 ? " (better)" : step.verdict === -1 ? " (worse)" : "";
  const verdictClass = step.verdict === 1 ? "verdict-good" : step.verdict === -1 ? "verdict-bad" : "";
  div.innerHTML = `<span class="${verdictClass}">${step.cmd_text}${verdictText}</span>`;
  const img = document.createElement("img");
  img.src = `data:image/png;base64,${step.image_b64}`;
  div.appendChild(img);
  el("messages").appendChild(div);
  el("messages").scrollTop = el("messages").scrollHeight;
}

async function loadState() {
  const r = await fetch("/api/state");
  await refresh(await r.json());
}

async function sendCommand(text) {
  const body = {
    text,
    parser: el("parser-select").value,
    search: el("search-toggle").checked,
    evaluator: el("evaluator-select").value,
    steps: parseInt(el("steps-input").value, 10),
  };
  const r = await fetch("/api/command", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) });
  if (r.status !== 200) {
    const err = await r.json();
    alert(`Could not parse: ${err.detail}`);
    return;
  }
  const data = await r.json();
  await refresh(data);
  if (!data.pending) appendMessage(data.current);
}

async function judge(verdict) {
  const r = await fetch("/api/judge", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ verdict }) });
  const data = await r.json();
  await refresh(data);
  if (!data.pending) appendMessage(data.current);
}

el("send-btn").addEventListener("click", () => {
  const text = el("text-input").value.trim();
  if (!text) return;
  el("text-input").value = "";
  sendCommand(text);
});
el("text-input").addEventListener("keydown", (e) => { if (e.key === "Enter") el("send-btn").click(); });
el("back-btn").addEventListener("click", async () => refresh(await (await fetch("/api/back", { method: "POST" })).json()));
el("forward-btn").addEventListener("click", async () => refresh(await (await fetch("/api/forward", { method: "POST" })).json()));
el("better-btn").addEventListener("click", () => judge("better"));
el("worse-btn").addEventListener("click", () => judge("worse"));
el("search-toggle").addEventListener("change", (e) => {
  el("evaluator-select").disabled = !e.target.checked;
  el("steps-input").disabled = !e.target.checked;
});

let mediaRecorder = null;
let chunks = [];
el("mic-btn").addEventListener("click", async () => {
  if (mediaRecorder && mediaRecorder.state === "recording") {
    mediaRecorder.stop();
    return;
  }
  const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
  mediaRecorder = new MediaRecorder(stream);
  chunks = [];
  mediaRecorder.ondataavailable = (e) => chunks.push(e.data);
  mediaRecorder.onstop = async () => {
    el("mic-btn").classList.remove("recording");
    stream.getTracks().forEach((t) => t.stop());
    const blob = new Blob(chunks, { type: "audio/webm" });
    const form = new FormData();
    form.append("audio", blob, "clip.webm");
    const r = await fetch("/api/transcribe", { method: "POST", body: form });
    const data = await r.json();
    if (data.text) sendCommand(data.text);
  };
  mediaRecorder.start();
  el("mic-btn").classList.add("recording");
});

loadState();
```
